# 'w' - WRITE (overwrites everything)
with open('test.txt', 'w') as f:
    f.write('New content')  # Old content is GONE!

# 'a' - APPEND (adds to end)
with open('test.txt', 'a') as f:
    f.write('Add this to the end')  # Keeps old content

# 'r' - READ
with open('test.txt', 'r') as f:
    content = f.read()  # Get all text

# 'x' - EXCLUSIVE CREATE (safety)
with open('new.txt', 'x') as f:
    f.write('Only creates if file NOT exists')

# 'r+' - READ AND WRITE
with open('test.txt', 'r+') as f:
    content = f.read()    # Can read
    f.write('New text')   # And write (overwrites from cursor position)
